isCorrectGuess(guess, actualWord):
    if guess is null or actualWord is null:
        return false
    return trim(guess) equals trim(actualWord), ignoring case


buildTruncatedSonnet(words, stopIndex):
    result = ""
    for i from 0 to stopIndex - 1:
        result = result + words[i] + " "
    result = result + "_____"
    return result
